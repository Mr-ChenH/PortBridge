cmake_minimum_required(VERSION 3.24)
project(PortBridgeWorkflowResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_QT6 ON CACHE BOOL "" FORCE)

target_include_directories(workflow_probe PRIVATE "${NODES_SOURCE}/examples/vertical_layout")
target_link_libraries(workflow_probe PRIVATE QtNodes::QtNodes Qt6::Widgets Qt6::Test Qt6::Network Qt6::WebSockets)
if(MINGW)
 target_compile_options(workflow_probe PRIVATE -finput-charset=UTF-8)
