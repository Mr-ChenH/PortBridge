cmake_minimum_required(VERSION 3.24)
project(PortBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network SerialPort Test)
find_package(Threads REQUIRED)
set(PORTBRIDGE_ASIO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/asio/asio" CACHE PATH "standalone Asio directory containing include/asio.hpp")
if(NOT EXISTS "${PORTBRIDGE_ASIO_ROOT}/include/asio.hpp")
  include(FetchContent)
  FetchContent_Declare(asio GIT_REPOSITORY https://github.com/chriskohlhoff/asio.git GIT_TAG 12e0ce9e0500bf0f247dbd1ae894272656456079 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(asio)
  set(PORTBRIDGE_ASIO_ROOT "${asio_SOURCE_DIR}/asio")
endif()
add_library(portbridge_network src/network/network_engine.cpp)
set_target_properties(portbridge_network PROPERTIES AUTOMOC OFF)
target_include_directories(portbridge_network PUBLIC include PRIVATE "${PORTBRIDGE_ASIO_ROOT}/include")
target_compile_definitions(portbridge_network PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(portbridge_network PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(portbridge_network PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(portbridge_network PUBLIC ws2_32 mswsock)
endif()
file(GLOB SESSION_SOURCES CONFIGURE_DEPENDS src/session/*.cpp src/session/*.hpp src/session/*.h)
add_library(portbridge_session ${SESSION_SOURCES} include/portbridge/session_controller.hpp)
target_include_directories(portbridge_session PUBLIC include PRIVATE src)
target_link_libraries(portbridge_session PUBLIC portbridge_network Qt6::Core Qt6::SerialPort)
file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.hpp src/ui/*.h)
add_library(portbridge_ui ${UI_SOURCES})
target_include_directories(portbridge_ui PUBLIC include src)
target_link_libraries(portbridge_ui PUBLIC portbridge_session Qt6::Widgets Qt6::Network)
add_executable(PortBridge WIN32 src/main.cpp assets/app-icon/app-icon.qrc)
set_target_properties(PortBridge PROPERTIES AUTORCC ON)
if(WIN32)
  enable_language(RC)
  configure_file(assets/app-icon/app-icon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/portbridge-icon.rc" @ONLY)
  target_sources(PortBridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/portbridge-icon.rc")
endif()
target_link_libraries(PortBridge PRIVATE portbridge_ui)
if(MINGW)
  foreach(target portbridge_network portbridge_session portbridge_ui PortBridge)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  endforeach()
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/network_bench.cpp")
  add_executable(portbridge_bench tools/network_bench.cpp)
  set_target_properties(portbridge_bench PROPERTIES AUTOMOC OFF)
  target_include_directories(portbridge_bench PRIVATE "${PORTBRIDGE_ASIO_ROOT}/include")
  target_compile_definitions(portbridge_bench PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED)
  target_link_libraries(portbridge_bench PRIVATE portbridge_network)
endif()
if(BUILD_TESTING)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_network.cpp")
    add_executable(test_network tests/test_network.cpp)
    set_target_properties(test_network PROPERTIES AUTOMOC OFF)
    target_include_directories(test_network PRIVATE "${PORTBRIDGE_ASIO_ROOT}/include")
    target_compile_definitions(test_network PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED)
    target_link_libraries(test_network PRIVATE portbridge_network)
    add_test(NAME network COMMAND test_network)
    set_tests_properties(network PROPERTIES TIMEOUT 45)
  endif()
  foreach(name session ui)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${name}.cpp")
      add_executable(test_${name} tests/test_${name}.cpp)
      target_link_libraries(test_${name} PRIVATE portbridge_${name} Qt6::Test)
      add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
        "-DTEST_EXECUTABLE=$<TARGET_FILE:test_${name}>"
        "-DTEST_LOG=${CMAKE_BINARY_DIR}/test-reports/${name}.txt"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/RunQtTest.cmake")
      set_tests_properties(${name} PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
  endforeach()
endif()
install(TARGETS PortBridge RUNTIME DESTINATION .)
