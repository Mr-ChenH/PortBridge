cmake_minimum_required(VERSION 3.24)
project(WorkflowIndependentReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Test SerialPort OpenGL WebSockets)
find_package(Threads REQUIRED)
# Link only worker C's genuine frozen protocol archive and static dependencies.
set(PROTOCOL "${ROOT}/build/workflow-protocol/native")
add_library(real_protocol INTERFACE)
target_link_libraries(real_protocol INTERFACE "${PROTOCOL}/libportbridge_workflow_protocol.a" "${PROTOCOL}/workflow-deps/cpr/cpr/libcpr.a" "${PROTOCOL}/workflow-deps/curl/lib/libcurl.a" "${PROTOCOL}/workflow-deps/cares/src/lib/libcares.a" "${ROOT}/.deps/workflow/openssl-install/lib64/libssl.a" "${ROOT}/.deps/workflow/openssl-install/lib64/libcrypto.a" advapi32 iphlpapi mswsock crypt32 bcrypt ws2_32 Qt6::Core)
set(BUILD_TESTING OFF)
set(BUILD_EXAMPLES OFF)
set(BUILD_DOCS OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static QtNodes for matching product ABI" FORCE)
set(BUILD_SHARED_LIBS OFF)
set(USE_QT6 ON)
add_subdirectory("${ROOT}/.deps/workflow-research/nodeeditor-7c6341a66a8e46b8988140b9e60d892b6a3560b3" qtnodes EXCLUDE_FROM_ALL)
add_library(review_network "${ROOT}/src/network/network_engine.cpp")
set_target_properties(review_network PROPERTIES AUTOMOC OFF)
target_include_directories(review_network PUBLIC "${ROOT}/include" PRIVATE "${ROOT}/.deps/asio/asio/include")
target_compile_definitions(review_network PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(review_network PUBLIC Threads::Threads ws2_32 mswsock)
file(GLOB SESSION "${ROOT}/src/session/*.cpp" "${ROOT}/src/session/*.hpp")
add_library(review_session ${SESSION} "${ROOT}/include/portbridge/session_controller.hpp")
target_include_directories(review_session PUBLIC "${ROOT}/include" PRIVATE "${ROOT}/src")
target_link_libraries(review_session PUBLIC review_network Qt6::Core Qt6::SerialPort)
file(GLOB CORE "${ROOT}/src/workflow/*.cpp" "${ROOT}/src/workflow/*.hpp")
add_library(review_core ${CORE} "${ROOT}/include/portbridge/workflow.hpp")
target_include_directories(review_core PUBLIC "${ROOT}/include" PRIVATE "${ROOT}/src/workflow")
target_link_libraries(review_core PUBLIC review_session real_protocol Qt6::Core)
file(GLOB UI "${ROOT}/src/ui/*.cpp" "${ROOT}/src/ui/*.hpp")
add_library(review_ui ${UI})
target_include_directories(review_ui PUBLIC "${ROOT}/include" "${ROOT}/src")
target_link_libraries(review_ui PUBLIC review_core QtNodes::QtNodes Qt6::Widgets Qt6::Network)
foreach(target review_network review_session review_core review_ui)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
endforeach()
foreach(name workflow workflow_ui workflow_integration)
  add_executable(test_${name} "${ROOT}/tests/test_${name}.cpp")
  target_link_libraries(test_${name} PRIVATE review_ui Qt6::Test Qt6::Network)
endforeach()
target_compile_definitions(test_workflow_ui PRIVATE PORTBRIDGE_WORKFLOW_REAL_PROTOCOL)
add_executable(test_workflow_e2e "${ROOT}/tests/test_workflow_e2e.cpp")
target_link_libraries(test_workflow_e2e PRIVATE review_ui Qt6::Test Qt6::WebSockets)
add_executable(independent independent.cpp)
target_link_libraries(independent PRIVATE review_ui Qt6::Test Qt6::WebSockets)
